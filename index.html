    <!-- Core App Logic -->

    <script>

        // --- SCENERY & ENGINE SETUP ---

        const container = document.getElementById('canvas-container');

        const scene = new THREE.Scene();

        scene.background = new THREE.Color(0x0b131e);

        scene.fog = new THREE.FogExp2(0x0b131e, 0.035);


        const camera = new THREE.PerspectiveCamera(65, window.innerWidth / window.innerHeight, 0.1, 1000);

        camera.position.set(0, 1.6, 5);


        const renderer = new THREE.WebGLRenderer({ antialias: true });

        renderer.setSize(window.innerWidth, window.innerHeight);

        renderer.setPixelRatio(window.devicePixelRatio);

        container.appendChild(renderer.domElement);


        // --- LIGHTING ---

        const ambientLight = new THREE.AmbientLight(0x17324D, 1.2);

        scene.add(ambientLight);


        const pointLight = new THREE.PointLight(0x287F83, 2, 15);

        pointLight.position.set(0, 3, 0);

        scene.add(pointLight);


        // --- ROOM ARCHITECTURE (Entrance Node) ---

        const roomGroup = new THREE.Group();

        scene.add(roomGroup);


        // Floor

        const floorGeo = new THREE.PlaneGeometry(10, 15);

        const floorMat = new THREE.MeshStandardMaterial({ color: 0x102A43, roughness: 0.8 });

        const floor = new THREE.Mesh(floorGeo, floorMat);

        floor.rotation.x = -Math.PI / 2;

        roomGroup.add(floor);


        // Walls

        const wallMat = new THREE.MeshStandardMaterial({ color: 0x17324D, roughness: 0.9 });

        const leftWall = new THREE.Mesh(new THREE.BoxGeometry(0.5, 4, 15), wallMat);

        leftWall.position.set(-5, 2, 0);

        roomGroup.add(leftWall);


        const rightWall = new THREE.Mesh(new THREE.BoxGeometry(0.5, 4, 15), wallMat);

        rightWall.position.set(5, 2, 0);

        roomGroup.add(rightWall);


        // --- INTERACTIVE LOCUS (The 2x2 Cell Core Object) ---

        const interactiveObjects = [];

        

        const locusGeo = new THREE.BoxGeometry(1.2, 1.2, 1.2);

        const locusMat = new THREE.MeshStandardMaterial({ color: 0x287F83, roughness: 0.3, emissive: 0x102A43 });

        const locusMesh = new THREE.Mesh(locusGeo, locusMat);

        locusMesh.position.set(0, 1, -4);

        locusMesh.userData = { 

            title: "True Positive / False Positive (2x2 Cell Core)", 

            category: "Wing A: Diagnostics Foyer",

            desc: "The foundation of diagnostic testing. Anchored as the entrance bridge guardrails, separating disease presence from diagnostic detection." 

        };

        scene.add(locusMesh);

        interactiveObjects.push(locusMesh);


        // Floating ambient particles for atmosphere

        const particleCount = 200;

        const particleGeo = new THREE.BufferGeometry();

        const particlePositions = new Float32Array(particleCount * 3);

        for(let i = 0; i < particleCount * 3; i += 3) {

            particlePositions[i] = (Math.random() - 0.5) * 20;

            particlePositions[i+1] = Math.random() * 5;

            particlePositions[i+2] = (Math.random() - 0.5) * 20;

        }

        particleGeo.setAttribute('position', new THREE.BufferAttribute(particlePositions, 3));

        const particleMat = new THREE.PointsMaterial({ color: 0x287F83, size: 0.05, transparent: true, opacity: 0.6 });

        const particles = new THREE.Points(particleGeo, particleMat);

        scene.add(particles);


        // --- RAYCASTING & CLICK INTERACTION ---

        const raycaster = new THREE.Raycaster();

        const mouse = new THREE.Vector2();


        window.addEventListener('click', (event) => {

            mouse.x = (event.clientX / window.innerWidth) * 2 - 1;

            mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;


            raycaster.setFromCamera(mouse, camera);

            const intersects = raycaster.intersectObjects(interactiveObjects);


            if (intersects.length > 0) {

                const data = intersects[0].object.userData;

                showModal(data.title, data.category, data.desc);

            }

        });


        function showModal(title, category, desc) {

            document.getElementById('modal-title').innerText = title;

            document.getElementById('modal-category').innerText = category;

            document.getElementById('modal-desc').innerText = desc;

            document.getElementById('modal').classList.remove('hidden');

            document.getElementById('modal').classList.add('flex');

        }


        function closeModal() {

            document.getElementById('modal').classList.add('hidden');

            document.getElementById('modal').classList.remove('flex');

        }


        // --- ANIMATION LOOP ---

        let clock = new THREE.Clock();

        function animate() {

            requestAnimationFrame(animate);

            let elapsedTime = clock.getElapsedTime();

            

            // Gentle floating/pulsing of the interactive locus

            locusMesh.rotation.y = elapsedTime * 0.5;

            locusMesh.position.y = 1 + Math.sin(elapsedTime * 2) * 0.1;


            renderer.render(scene, camera);

        }

        animate();


        // --- RESIZE HANDLER ---

        window.addEventListener('resize', () => {

            camera.aspect = window.innerWidth / window.innerHeight;

            camera.updateProjectionMatrix();

            renderer.setSize(window.innerWidth, window.innerHeight);

        });

    </script>
